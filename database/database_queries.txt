-- ============================================================
-- SQL DDL and DML Queries for ETCMS Application
-- ============================================================

-- ------------------------------------------------------------
-- 1. DATA DEFINITION LANGUAGE (DDL) - TABLE CREATION
-- ------------------------------------------------------------

-- Table: designations
CREATE TABLE designations (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL
);

-- Table: employees
CREATE TABLE employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    designationId INT NOT NULL,
    phone VARCHAR(50),
    dateOfJoining DATE,
    FOREIGN KEY (designationId) REFERENCES designations(id)
);

-- Table: users
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    employeeId INT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL,
    lastLogin DATETIME NULL,
    FOREIGN KEY (employeeId) REFERENCES employees(id)
);

-- Table: uploadedCertificates
CREATE TABLE uploadedCertificates (
    id INT PRIMARY KEY AUTO_INCREMENT,
    employeeId INT NOT NULL,
    certificateName VARCHAR(255) NOT NULL,
    organization VARCHAR(255) NOT NULL,
    issueDate DATE,
    expiryDate DATE NULL,
    certificateId VARCHAR(100) NULL,
    verificationUrl VARCHAR(500) NULL,
    fileName VARCHAR(255),
    fileType VARCHAR(100),
    fileData BLOB NULL,
    status VARCHAR(50) DEFAULT 'PENDING',
    rejectionReason TEXT NULL,
    uploadedAt DATETIME,
    verifiedAt DATETIME NULL,
    verifiedBy VARCHAR(255) NULL,
    FOREIGN KEY (employeeId) REFERENCES employees(id)
);

-- Table: adminAssessments
CREATE TABLE adminAssessments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    courseName VARCHAR(255),
    assessmentDate DATE,
    maximumMarks INT,
    createdAt DATETIME
);

-- Table: assessment_assigned_employees
-- (Join table to handle the many-to-many relationship of assignedEmployeeIds)
CREATE TABLE assessment_assigned_employees (
    assessmentId INT NOT NULL,
    employeeId INT NOT NULL,
    PRIMARY KEY (assessmentId, employeeId),
    FOREIGN KEY (assessmentId) REFERENCES adminAssessments(id),
    FOREIGN KEY (employeeId) REFERENCES employees(id)
);

-- Table: assessmentResults
CREATE TABLE assessmentResults (
    id INT PRIMARY KEY AUTO_INCREMENT,
    assessmentId INT NOT NULL,
    employeeId INT NOT NULL,
    marks INT NULL,
    remarks TEXT NULL,
    status VARCHAR(50) DEFAULT 'PENDING',
    FOREIGN KEY (assessmentId) REFERENCES adminAssessments(id),
    FOREIGN KEY (employeeId) REFERENCES employees(id)
);
